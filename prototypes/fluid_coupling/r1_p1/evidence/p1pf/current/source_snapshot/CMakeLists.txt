cmake_minimum_required(VERSION 3.16)
project(judas_r1_p1 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
add_executable(judas_r1_p1 main.cpp)
target_compile_options(judas_r1_p1 PRIVATE -Wall -Wextra -Wpedantic -O2)

# Isolated optional frozen-geometry component; the original transport target
# has no LAPACK dependency and retains its existing compilation settings.
option(JUDAS_BUILD_P1PF "Build isolated P1-PF projection component" OFF)
if(JUDAS_BUILD_P1PF)
    find_library(P1PF_LAPACK NAMES lapack liblapack.so.3)
    if(NOT P1PF_LAPACK)
        message(FATAL_ERROR "P1-PF requires an LP64 LAPACK library exposing dgesvd_")
    endif()
    add_executable(judas_r1_p1pf main.cpp)
    target_compile_definitions(judas_r1_p1pf PRIVATE JUDAS_P1PF=1)
    target_compile_options(judas_r1_p1pf PRIVATE -Wall -Wextra -Wpedantic -O2)
    target_link_libraries(judas_r1_p1pf PRIVATE ${P1PF_LAPACK})
    message(STATUS "P1-PF LAPACK: ${P1PF_LAPACK}")
endif()
